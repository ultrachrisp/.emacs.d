*** Eglot
Built-in LSP client. It reports through Flymake, navigates with xref, shows
docs with eldoc and completes through completion-at-point (Corfu).

For JS/TS, the servers are picked per project: TypeScript 7's own
=tsc --lsp=, plus ESLint, oxlint and Tailwind when the project uses them.
Several servers run behind one connection through Rassumfrassum (=rass=).
Install: =uv tool install rassumfrassum= and
=npm i -g vscode-langservers-extracted @tailwindcss/language-server=.
#+BEGIN_SRC emacs-lisp
(defun my/eglot-node-bin (root name)
  "Return NAME from ROOT's node_modules/.bin if it exists."
  (let ((bin (expand-file-name (concat "node_modules/.bin/" name) root)))
    (and (file-executable-p bin) bin)))

(defun my/eglot-typescript-major (root)
  "Return the major version of the TypeScript installed in ROOT, or 0."
  (let ((pkg (expand-file-name "node_modules/typescript/package.json" root)))
    (if (file-readable-p pkg)
        (with-temp-buffer
          (insert-file-contents pkg)
          (truncate (string-to-number (gethash "version" (json-parse-buffer) "0"))))
      0)))

(defun my/eglot-js-servers (root)
  "Return the language server commands for the JS/TS project at ROOT.
TypeScript 7+ ships its server as `tsc --lsp'; older projects use the
global TypeScript 7."
  (delq nil
        (list (list (or (and (>= (my/eglot-typescript-major root) 7)
                             (my/eglot-node-bin root "tsc"))
                        "tsc")
                    "--lsp" "--stdio")
              (when (directory-files root nil "\\`\\(eslint\\.config\\.[cm]?[jt]s\\|\\.eslintrc.*\\)\\'")
                '("vscode-eslint-language-server" "--stdio"))
              (when-let* ((oxlint (my/eglot-node-bin root "oxlint")))
                (list oxlint "--lsp"))
              (when (file-directory-p (expand-file-name "node_modules/tailwindcss" root))
                '("tailwindcss-language-server" "--stdio")))))

(defconst my/rass-logic-dir (expand-file-name "rass" user-emacs-directory)
  "Directory holding emacs_rass.py, the logic class rass runs with.
It answers ESLint's settings request and works around a rass bug that
hides TypeScript 7's diagnostics; see the file for details.")

(defun my/eglot-js-contact (_interactive project)
  "Eglot contact for JS/TS: a single server directly, several through rass."
  (let ((servers (my/eglot-js-servers
                  (expand-file-name (if project (project-root project) default-directory)))))
    (if (cdr servers)
        (append (list "env" (concat "PYTHONPATH=" my/rass-logic-dir)
                      "rass" "--logic-class" "emacs_rass.EmacsTypeScriptLogic")
                (mapcan (lambda (server) (cons "--" (copy-sequence server))) servers))
      (car servers))))

(defun my/eglot-organize-imports-on-save ()
  "Organize imports before saving, when the server offers it."
  (add-hook 'before-save-hook
            (lambda ()
              (ignore-errors
                (eglot-code-action-organize-imports (point-min) (point-max))))
            nil t))

(use-package eglot
    :straight (:type built-in)
    :hook (((typescript-ts-mode tsx-ts-mode js-ts-mode
             json-ts-mode css-ts-mode scss-mode mhtml-ts-mode) . eglot-ensure)
           ((typescript-ts-mode tsx-ts-mode js-ts-mode) . my/eglot-organize-imports-on-save))
    :bind (:map eglot-mode-map
                ("C-c l a" . eglot-code-actions)
                ("C-c l r" . eglot-rename)
                ("C-c l o" . eglot-code-action-organize-imports)
                ("C-c l f" . eglot-format)
                ("C-c l d" . eldoc-doc-buffer))
    :custom
    (eglot-autoshutdown t)
    (eglot-extend-to-xref t)
    (eglot-events-buffer-config '(:size 0 :format full))
    :config
    (add-to-list 'eglot-server-programs
                 '(((js-ts-mode :language-id "javascript")
                    (tsx-ts-mode :language-id "typescriptreact")
                    (typescript-ts-mode :language-id "typescript"))
                   . my/eglot-js-contact))
    (setq-default eglot-workspace-configuration
                  (let ((hints '(:parameterNames (:enabled "all")
                                 :variableTypes (:enabled t)
                                 :propertyDeclarationTypes (:enabled t)
                                 :functionLikeReturnTypes (:enabled t)
                                 :enumMemberValues (:enabled t))))
                    `(:typescript (:inlayHints ,hints)
                      :javascript (:inlayHints ,hints)))))
