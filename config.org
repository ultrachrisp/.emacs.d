*** Themes
The built-in Modus themes, following the OS light/dark setting. On macOS,
emacs-plus reports appearance changes through
=ns-system-appearance-change-functions=, instantly and without polling. On
Linux/WSL, =auto-dark= follows the desktop setting (D-Bus on Linux, Windows
theme via PowerShell on WSL); switch by hand with
=M-x auto-dark-toggle-appearance=.
#+BEGIN_SRC emacs-lisp
(defconst my/themes '((dark . modus-vivendi) (light . modus-operandi-tinted))
  "Theme to use for each OS appearance.")

(defun my/apply-appearance (appearance)
  "Load the theme in `my/themes' for APPEARANCE (`light' or `dark').
APPEARANCE is nil without a GUI frame (e.g. `emacs -nw'); use dark then."
  (mapc #'disable-theme custom-enabled-themes)
  (load-theme (alist-get (or appearance 'dark) my/themes) t))

(when (boundp 'ns-system-appearance-change-functions)
  (add-hook 'ns-system-appearance-change-functions #'my/apply-appearance)
  (my/apply-appearance ns-system-appearance))

;; Installed everywhere (so the lockfile pins it), enabled only without the
;; emacs-plus hook.
(use-package auto-dark
    :custom
    ;; ((dark themes) (light themes))
    (auto-dark-themes `((,(alist-get 'dark my/themes))
                        (,(alist-get 'light my/themes))))
    (auto-dark-allow-powershell t)
    ;; WSL polls by running powershell.exe synchronously; poll rarely so
    ;; Emacs doesn't stall. Unused with D-Bus or emacs-plus.
    (auto-dark-polling-interval-seconds 60)
    :config
    (unless (boundp 'ns-system-appearance-change-functions)
      (if (display-graphic-p)
          (auto-dark-mode 1)
        (my/apply-appearance 'dark))))
#+END_SRC

*** Eglot
Built-in LSP client. It reports through Flymake, navigates with xref, shows
docs with eldoc and completes through completion-at-point (Corfu).

For JS/TS, the servers are picked per project: TypeScript 7's own
=tsc --lsp=, plus ESLint, oxlint and Tailwind when the project uses them.
Several servers run behind one connection through Rassumfrassum (=rass=).
Vue SFCs (=my/vue-mode=) run =vue-language-server= and =vtsls= (with
=@vue/typescript-plugin=) through rass, plus ESLint, oxlint and Tailwind when
the project uses them; =rass/emacs_rass.py= relays the Vue server's TypeScript requests to vtsls.
Install: =uv tool install rassumfrassum=, and the language servers through
mise (see External Dependencies).

Rust uses =rust-analyzer=, checking with clippy. Python uses Eglot's
built-in =rass python=, which runs ty (types) and ruff (lint) together.
Install: =rustup component add rust-analyzer clippy rustfmt= and
=uv tool install ty ruff=.
#+BEGIN_SRC emacs-lisp
(defun my/eglot-node-bin (root name)
  "Return NAME from ROOT's node_modules/.bin if it exists."
  (let ((bin (expand-file-name (concat "node_modules/.bin/" name) root)))
    (and (file-executable-p bin) bin)))

(defun my/eglot-typescript-major (root)
  "Return the major version of the TypeScript installed in ROOT, or 0."
  (let ((pkg (expand-file-name "node_modules/typescript/package.json" root)))
    (if (file-readable-p pkg)
        (with-temp-buffer
          (insert-file-contents pkg)
          (truncate (string-to-number (gethash "version" (json-parse-buffer) "0"))))
      0)))

(defun my/eglot-web-extras (root)
  "Return optional web language servers used by the project at ROOT."
  (delq nil
        (list (when (directory-files root nil "\\`\\(eslint\\.config\\.[cm]?[jt]s\\|\\.eslintrc.*\\)\\'")
                '("vscode-eslint-language-server" "--stdio"))
              (when-let* ((oxlint (my/eglot-node-bin root "oxlint")))
                (list oxlint "--lsp"))
              (when (file-directory-p (expand-file-name "node_modules/tailwindcss" root))
                '("tailwindcss-language-server" "--stdio")))))

(defun my/eglot-js-servers (root)
  "Return the language server commands for the JS/TS project at ROOT.
TypeScript 7+ ships its server as `tsc --lsp'; older projects use the
global TypeScript 7."
  (cons (list (or (and (>= (my/eglot-typescript-major root) 7)
                       (my/eglot-node-bin root "tsc"))
                  "tsc")
              "--lsp" "--stdio")
        (my/eglot-web-extras root)))

(defconst my/rass-logic-dir (expand-file-name "rass" user-emacs-directory)
  "Directory holding emacs_rass.py, the logic class rass runs with.
It answers ESLint's settings request and works around a rass bug that
hides TypeScript 7's diagnostics; see the file for details.")

(defun my/eglot-js-contact (_interactive project)
  "Eglot contact for JS/TS: a single server directly, several through rass."
  (let ((servers (my/eglot-js-servers
                  (expand-file-name (if project (project-root project) default-directory)))))
    (if (cdr servers)
        (append (list "env" (concat "PYTHONPATH=" my/rass-logic-dir)
                      "rass" "--logic-class" "emacs_rass.EmacsTypeScriptLogic")
                (mapcan (lambda (server) (cons "--" (copy-sequence server))) servers))
      (car servers))))

(defun my/global-npm-package-dir (package bin)
  "Return the install directory of the global npm PACKAGE, which provides BIN.
Asks mise first; otherwise follows BIN back to its package (npm -g layout)."
  (or (when-let* ((mise (executable-find "mise"))
                  (where (car (process-lines-ignore-status mise "where" (concat "npm:" package))))
                  (dir (expand-file-name (concat "node_modules/" package) where))
                  ((file-directory-p dir)))
        (file-truename dir))
      (when-let* ((path (executable-find bin))
                  ((not (string-match-p "/mise/shims/" path)))
                  (dir (locate-dominating-file (file-truename path) "package.json")))
        (expand-file-name dir))))

(defun my/node-module-dir (from module)
  "Return the absolute directory of MODULE as Node resolves it from FROM."
  (when-let* ((from from)
              (node (executable-find "node"))
              (main (car (process-lines-ignore-status
                          node "-e" (format "console.log(require.resolve(%S, {paths: [%S]}))"
                                            module (expand-file-name from)))))
              ((file-exists-p main))
              (dir (locate-dominating-file main "package.json")))
    (expand-file-name dir)))

(defun my/vue-global-module (name)
  "Return the directory of NAME as used by the global vue-language-server."
  (my/node-module-dir (my/global-npm-package-dir "@vue/language-server" "vue-language-server")
                      name))

(defun my/vue-tsdk (root)
  "Return a TypeScript lib directory for Vue: the project's, else vtsls's."
  (seq-find #'file-directory-p
            (list (expand-file-name "node_modules/typescript/lib" root)
                  (when-let* ((ts (my/node-module-dir
                                   (my/global-npm-package-dir "@vtsls/language-server" "vtsls")
                                   "typescript")))
                    (expand-file-name "lib" ts)))))

(defvar my/vue-ts-plugin-dir nil
  "Cached absolute path to `@vue/typescript-plugin', resolved lazily.")

(defun my/eglot-workspace-config (server)
  "Return Eglot workspace configuration for SERVER.
Resolves `@vue/typescript-plugin' lazily so non-Vue servers don't spawn
`mise' and `node' on Eglot load."
  (let* ((hints '(:parameterNames (:enabled "all")
                  :variableTypes (:enabled t)
                  :propertyDeclarationTypes (:enabled t)
                  :functionLikeReturnTypes (:enabled t)
                  :enumMemberValues (:enabled t)))
         (base `(:typescript (:inlayHints ,hints)
                 :javascript (:inlayHints ,hints)
                 :rust-analyzer (:check (:command "clippy")))))
    (if (and server (memq 'my/vue-mode (eglot--major-modes server)))
        (let ((plugin (or my/vue-ts-plugin-dir
                          (setq my/vue-ts-plugin-dir
                                (my/vue-global-module "@vue/typescript-plugin")))))
          (append base
                  `(:vtsls (:tsserver
                            (:globalPlugins
                             [(:name "@vue/typescript-plugin"
                               :location ,(or plugin "")
                               :languages ["vue"]
                               :configNamespace "typescript"
                               :enableForWorkspaceTypeScriptVersions t)])))))
      base)))

(defun my/eglot-vue-contact (_interactive project)
  "Eglot contact for Vue SFCs: vue-language-server and vtsls through rass.
vtsls loads @vue/typescript-plugin (see `my/eglot-workspace-config'),
and emacs_rass.py forwards the Vue server's TypeScript requests to it.
ESLint, oxlint, and Tailwind join when the project uses them."
  (let* ((root (expand-file-name (if project (project-root project) default-directory)))
         (tsdk (my/vue-tsdk root)))
    `("env" ,(concat "PYTHONPATH=" my/rass-logic-dir)
      "rass" "--logic-class" "emacs_rass.EmacsVueLogic"
      "--" "vue-language-server" "--stdio"
      "--" "vtsls" "--stdio"
      ,@(mapcan (lambda (server) (cons "--" (copy-sequence server)))
                (my/eglot-web-extras root))
      ,@(when tsdk `(:initializationOptions (:typescript (:tsdk ,tsdk)))))))

*** Agent Shell (Antigravity & Claude Code)
Native Emacs interface to Google Antigravity (=agy acp=) and Claude Code
(=claude-agent-acp=) over the Agent Client Protocol, built on =comint-mode=,
=project.el=, and =completion-at-point=. Docks in a right-hand side window
(jump with =C-x o=), integrates with =project.el= (=C-x p a= / =C-c p a=),
and disables automatic Corfu popups for prose while keeping =@file= and
=/command= completion active.

Use =C-c a a= / =C-c a n= to pick between installed agents (Antigravity
preselected), =C-c a g= or =C-c a C= to launch Antigravity or Claude Code
directly, and =C-c a b= to switch between running agent sessions in the side
window. With multiple agents open in a project, =C-u C-u C-c a s= or
=C-u C-c a d= prompts for which agent should receive the context.
#+BEGIN_SRC emacs-lisp
(defun my/agent-shell-setup ()
  "Keep Corfu from auto-popping on prose while preserving `@' and `/' completion."
  (setq-local corfu-auto nil
              corfu-quit-at-boundary t
              corfu-quit-no-match t))

(defun my/agent-shell-configs ()
  "Return Antigravity and Claude Code makers, filtered to installed CLIs."
  (let* ((makers (list #'agent-shell-antigravity-make-agent-config
                       #'agent-shell-anthropic-make-claude-code-config))
         (available (seq-filter
                     (lambda (maker)
                       (when-let* ((cfg (funcall maker))
                                   (client-maker (map-elt cfg :client-maker))
                                   (client (ignore-errors
                                             (funcall client-maker (current-buffer))))
                                   (command (map-elt client :command)))
                         (executable-find command)))
                     makers)))
    (or available makers)))

(defun my/agent-shell-switch-buffer ()
  "Switch to an `agent-shell' buffer in the configured side window."
  (interactive)
  (require 'agent-shell)
  (let ((shell-buffer (agent-shell--read-shell-buffer
                       :prompt "Switch to agent-shell buffer: ")))
    (agent-shell--display-buffer shell-buffer)))

(defun my/agent-shell-send-flymake-diagnostics (&optional pick-shell)
  "Insert all Flymake diagnostics for the current buffer into `agent-shell'.
With prefix argument PICK-SHELL, prompt for which agent buffer to send to."
  (interactive "P")
  (require 'flymake)
  (let ((diags (and (bound-and-true-p flymake-mode)
                    (flymake-diagnostics))))
    (unless diags
      (user-error "No Flymake diagnostics in current buffer"))
    (let* ((project (project-current))
           (root (and project (project-root project)))
           (file (cond
                  ((and buffer-file-name root)
                   (file-relative-name buffer-file-name root))
                  (buffer-file-name)
                  (t (buffer-name))))
           (lines (mapconcat
                   (lambda (d)
                     (format "- %s:%d [%s] %s"
                             file
                             (line-number-at-pos (flymake-diagnostic-beg d) t)
                             (flymake-diagnostic-type d)
                             (flymake-diagnostic-text d)))
                   diags "\n")))
      (require 'agent-shell)
      (let ((shell-buffer (when pick-shell
                            (agent-shell--read-shell-buffer
                             :prompt "Send Flymake diagnostics to: "))))
        (agent-shell-insert
         :text (format "Fix these Flymake diagnostics in %s:\n%s\n" file lines)
         :shell-buffer shell-buffer)))))

(use-package agent-shell
    :bind (("C-c a a" . agent-shell)
           ("C-c a t" . agent-shell-toggle)
           ("C-c a n" . agent-shell-new-shell)
           ("C-c a g" . agent-shell-antigravity-start-agent)
           ("C-c a C" . agent-shell-anthropic-start-claude-code)
           ("C-c a b" . my/agent-shell-switch-buffer)
           ("C-c a s" . agent-shell-send-dwim)
           ("C-c a r" . agent-shell-send-region)
           ("C-c a f" . agent-shell-send-current-file)
           ("C-c a F" . agent-shell-send-file)
           ("C-c a c" . agent-shell-prompt-compose)
           ("C-c a !" . agent-shell-insert-shell-command-output)
           ("C-c a d" . my/agent-shell-send-flymake-diagnostics))
    :hook (agent-shell-mode . my/agent-shell-setup)
    :custom
    (agent-shell-agent-configs #'my/agent-shell-configs)
    (agent-shell-preferred-agent-config '(preselect . antigravity))
    (agent-shell-antigravity-default-config-options
     '(("mode" . "auto")
       ("thought_level" . "medium")))
    (agent-shell-anthropic-default-session-mode-id "acceptEdits")
    ;; Avoid `(dedicated . t)' so Antigravity and Claude Code share the side window.
    (agent-shell-display-action
     '(display-buffer-in-side-window
       (side . right)
       (slot . 0)
       (window-width . 0.4)
       (window-parameters . ((no-delete-other-windows . t)))))
    ;; Open clicked file links in a main window instead of replacing the agent pane.
    (agent-shell-file-display-action
     '(display-buffer-use-some-window
       (inhibit-same-window . t)))
    :init
    (with-eval-after-load 'project
      (keymap-set project-prefix-map "a" #'agent-shell)
      (add-to-list 'project-switch-commands '(agent-shell "Agent shell" ?a) t))
    (with-eval-after-load 'which-key
      (which-key-add-key-based-replacements "C-c a" "agent")))

** JavaScript, TypeScript & Vue
[[https://mise.jdx.dev][mise]] manages Node and the global language servers. Each tool is installed
on its own, so it survives Node upgrades, and projects still get their own
Node version from =.nvmrc= / =.node-version=.
#+BEGIN_SRC sh
brew install mise   # Linux: curl https://mise.run | sh
mkdir -p ~/.config/mise
cat > ~/.config/mise/config.toml <<'TOML'
[tools]
node = "26"
"npm:typescript" = "7"
"npm:vscode-langservers-extracted" = "latest"
"npm:@tailwindcss/language-server" = "latest"
"npm:@vue/language-server" = "latest"
"npm:@vtsls/language-server" = "latest"
TOML
mise install
#+END_SRC
- =typescript@7= provides =tsc --lsp=, the TypeScript server for projects
  whose own TypeScript is older than 7.
- =vscode-langservers-extracted= provides the ESLint, JSON, CSS and HTML servers.
- =@vue/language-server= + =@vtsls/language-server= serve =.vue= files.
- =consult-ripgrep= and =consult-fd= need =ripgrep= and =fd= (see Base tools).
- Prettier and oxlint come from each project's own =node_modules=.
